cmake_minimum_required(VERSION 3.20.0)

# Provides the SSD1306 OLED (with the exact init parameters this specific
# panel needs - segment-remap/com-invdir/multiplex-ratio etc, easy to get
# wrong by hand) and the D1 button as a proper gpio-keys node. Set here
# (before find_package) so plain `west build` picks it up with no extra
# flag needed, same reasoning as the self-contained CDC-ACM console setup.
# Does not cover the buzzer - that pinctrl/PWM setup stays in
# boards/xiao_rp2040.overlay, since the shield doesn't know about it.
set(SHIELD seeed_xiao_expansion_board)

find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(xiao_rp2040_mplay)

target_sources(app PRIVATE src/main.c src/mod_player.c src/vu_neopixel.c src/oled_display.c src/spectrum.c src/diag.c)

# Embed the playlist as flash data, generated at build time into the build
# directory rather than committing ~6x larger generated headers. Each MOD
# becomes a byte-list .inc (Zephyr's generate_inc_file_for_target()), and
# mod_table.h wraps them into mod_table[] in this order - the D1 button
# steps through it (see main.c). Add/remove/reorder songs here.
set(MOD_FILES
        AXEL_F.MOD
        POPCORN1.MOD
        ELYSIUM.MOD
        MOBY_DRAGONSFUNK.MOD
        12TH.MOD
        STEEL_CHAMBERS_2.MOD
        2010.MOD
        JOHN.MOD
        LISBONFE.MOD
)
set(MOD_GEN_DIR ${CMAKE_CURRENT_BINARY_DIR}/mods)
set(MOD_ARRAYS "")
set(MOD_ENTRIES "")
set(MOD_INDEX 0)
foreach(MOD_FILE ${MOD_FILES})
        generate_inc_file_for_target(app ${CMAKE_CURRENT_LIST_DIR}/mods/${MOD_FILE}
                                     ${MOD_GEN_DIR}/mod_${MOD_INDEX}.inc)
        string(APPEND MOD_ARRAYS
                "static const uint8_t mod_data_${MOD_INDEX}[] = {\n#include \"mod_${MOD_INDEX}.inc\"\n};\n")
        string(APPEND MOD_ENTRIES
                "\t{ \"${MOD_FILE}\", mod_data_${MOD_INDEX}, sizeof(mod_data_${MOD_INDEX}) },\n")
        math(EXPR MOD_INDEX "${MOD_INDEX} + 1")
endforeach()
file(GENERATE OUTPUT ${MOD_GEN_DIR}/mod_table.h CONTENT
"/* Generated by CMakeLists.txt from MOD_FILES - do not edit. */
#include <stddef.h>
#include <stdint.h>

struct mod_entry {
\tconst char *name;
\tconst uint8_t *data;
\tsize_t len;
};

${MOD_ARRAYS}
static const struct mod_entry mod_table[] = {
${MOD_ENTRIES}};
")
target_include_directories(app PRIVATE ${MOD_GEN_DIR})
